Notes de livraison - script a executer sur la base de recette

CREATE TABLE customers (
  customer_id NUMBER(10) PRIMARY KEY,
  name        VARCHAR2(100) NOT NULL,
  email       VARCHAR2(200),
  created_at  DATE DEFAULT SYSDATE NOT NULL
);

CREATE TABLE orders (
  order_id    NUMBER(10) NOT NULL,
  customer_id NUMBER(10) NOT NULL,
  total       NUMBER(12,2) DEFAULT 0,
  status      VARCHAR2(20),
  CONSTRAINT pk_orders PRIMARY KEY (order_id),
  CONSTRAINT fk_orders_cust FOREIGN KEY (customer_id) REFERENCES customers (customer_id)
);

CREATE OR REPLACE TRIGGER trg_orders_bi
BEFORE INSERT ON orders
FOR EACH ROW
BEGIN
  IF :NEW.status IS NULL THEN
    :NEW.status := 'NEW';
  END IF;
END;
/

CREATE OR REPLACE PROCEDURE purge_old(p_days IN NUMBER) IS
BEGIN
  DELETE FROM orders WHERE created_at < SYSDATE - p_days;
END;
/

DECLARE
  l_count NUMBER;
BEGIN
  SELECT COUNT(*) INTO l_count FROM orders;
  DBMS_OUTPUT.put_line('orders: ' || l_count);
END;
/

SELECT * FROM orders;
